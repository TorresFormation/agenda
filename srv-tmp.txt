var TF_BOOL = { actif: 1, theorie: 1, pratique: 1, test: 1, tout: 1, bloque: 1, rec: 1, own: 1 }, TF_LIST = { jours: 1 };

function installer() {
  var P = PropertiesService.getScriptProperties(), ss;
  if (P.getProperty('SS_ID')) ss = SpreadsheetApp.openById(P.getProperty('SS_ID'));
  else { ss = SpreadsheetApp.create('Torres Formation — Agenda (données)'); P.setProperty('SS_ID', ss.getId()); }
  P.deleteProperty('SCHEMA'); tfEnsureSchema_(ss, P);
  if (!P.getProperty('ADMIN_KEY')) P.setProperty('ADMIN_KEY', tfToken_() + tfToken_());
  ScriptApp.getProjectTriggers().forEach(function (t) { if (t.getHandlerFunction() === 'tfSync') ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('tfSync').timeBased().everyMinutes(15).create();
  CalendarApp.getDefaultCalendar().getName(); MailApp.getRemainingDailyQuota();
  tfSync();
  Logger.log('Installation terminée. Déployez maintenant l\'application Web.');
}

/* Crée / met à niveau les feuilles de données (ajoute les nouvelles colonnes sans toucher aux données) */
function tfEnsureSchema_(ss, P) {
  if (P.getProperty('SCHEMA') === '7') return;
  Object.keys(TF_TABLES).forEach(function (n) {
    var cols = TF_TABLES[n], sh = ss.getSheetByName(n) || ss.insertSheet(n);
    if (sh.getMaxColumns() < cols.length) sh.insertColumnsAfter(sh.getMaxColumns(), cols.length - sh.getMaxColumns());
    sh.getRange(1, 1, sh.getMaxRows(), sh.getMaxColumns()).setNumberFormat('@');
    sh.getRange(1, 1, 1, cols.length).setValues([cols]).setFontWeight('bold'); sh.setFrozenRows(1);
  });
  var d = ss.getSheetByName('Feuille 1') || ss.getSheetByName('Sheet1'); if (d && ss.getSheets().length > 1) ss.deleteSheet(d);
  P.setProperty('SCHEMA', '7');
}

function doGet(e) {
  var q = (e && e.parameter) || {}, t = HtmlService.createTemplateFromFile('Index');
  t.boot = JSON.stringify({ key: String(q.admin || q.c || ''), mode: q.admin ? 'admin' : (q.c ? 'client' : 'none'), contact: TF_CONFIG.contact }).replace(/</g, '\\x3c');
  return t.evaluate().setTitle('Torres Formation — Agenda').addMetaTag('viewport', 'width=device-width, initial-scale=1')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.DEFAULT);
}

/* Appelé par la page publique (GitHub Pages) */
function doPost(e) {
  var r = { a: '', k: '', p: {} }, out;
  try { r = JSON.parse(e.postData.contents); } catch (x) { }
  try { out = api(String(r.a || ''), String(r.k || ''), JSON.stringify(r.p || {})); }
  catch (err) { out = JSON.stringify({ ok: false, error: 'Service momentanément indisponible, réessayez.' }); }
  return ContentService.createTextOutput(out).setMimeType(ContentService.MimeType.JSON);
}

function api(action, key, json) {
  var P = PropertiesService.getScriptProperties(), res;
  var env = tfEnv_(P);
  if (action === 'lienAdmin') { // envoi du lien formateur à sa propre adresse, au plus 1 fois / 10 min
    var last = Number(P.getProperty('LAST_ADMIN_MAIL') || 0);
    if (Date.now() - last < 600000) return JSON.stringify({ ok: true, data: {} });
    P.setProperty('LAST_ADMIN_MAIL', String(Date.now()));
  }
  if (action === 'sante' || action === 'sauvegarder') { // contrôle de santé + sauvegarde immédiate (formateur seulement)
    if (String(key) !== env.adminKey) return JSON.stringify({ ok: false, error: 'Accès refusé.' });
    if (action === 'sauvegarder') { var lk = LockService.getScriptLock(); lk.waitLock(25000); try { P.deleteProperty('BACKUP_DAY'); tfBackup_(SpreadsheetApp.openById(P.getProperty('SS_ID')), P); } catch (x) { P.setProperty('BACKUP_ERROR', String(x && x.message || x)); } finally { lk.releaseLock(); } }
    return JSON.stringify({ ok: true, data: tfSante_(P, env) });
  }
  if (action === 'sync') { if (String(key) !== env.adminKey) return JSON.stringify({ ok: false, error: 'Accès refusé.' }); tfSync(); action = 'init'; }
  var evtMsg = '';
  if (action === 'evtMaj' || action === 'evtCreer' || action === 'evtSuppr') { // modification directe de Google Agenda (formateur seulement)
    if (String(key) !== env.adminKey) return JSON.stringify({ ok: false, error: 'Accès refusé.' });
    try { evtMsg = tfEvt_(action, JSON.parse(json || '{}')); } catch (x) { return JSON.stringify({ ok: false, error: String(x && x.message || x) }); }
    tfSync(); action = 'init';
  }
  var lock = LockService.getScriptLock(), write = action !== 'init';
  if (write) lock.waitLock(25000);
  try {
    var ss = SpreadsheetApp.openById(P.getProperty('SS_ID'));
    tfEnsureSchema_(ss, P);
    var db = tfLoad_(ss, P);
    res = tfApi(db, env, action, key, JSON.parse(json || '{}'));
    if (res.ok) {
      Object.keys(res.dirty || {}).forEach(function (n) { tfSave_(ss, n, n === 'ouverts' ? db.ouverts.map(function (j) { return { jour: j }; }) : db[n]); });
      if (res.mails && res.mails.length) tfQueueMails_(ss, res.mails);
    }
  } finally { if (write) { SpreadsheetApp.flush(); lock.releaseLock(); } }
  if (res.ok && res.sync) { tfSync(); var r2 = tfApi(tfLoad_(SpreadsheetApp.openById(P.getProperty('SS_ID')), P), env, 'init', key, {}); if (r2.ok) { r2.data.msg = res.data.msg; res.data = r2.data; } }
  else if (res.ok && res.mails && res.mails.length) { try { tfFlushMails_(); } catch (err) { } }
  if (res.ok && evtMsg) res.data.msg = evtMsg;
  if (res.ok && action === 'mensuel' && res.data && res.data.envoyes !== undefined && JSON.parse(json || '{}').mode === 'envoi') P.setProperty('LAST_MENSUEL', JSON.parse(json || '{}').ym || env.today.slice(0, 7));
  return JSON.stringify(res.ok ? { ok: true, data: res.data } : { ok: false, error: res.error });
}

function tfEnv_(P) {
  var now = new Date();
  return { today: Utilities.formatDate(now, TF_CONFIG.tz, 'yyyy-MM-dd'), now: now.toISOString(), minDelai: TF_CONFIG.minDelai, horizon: TF_CONFIG.horizon,
    adminKey: P.getProperty('ADMIN_KEY') || '', adminEmail: Session.getEffectiveUser().getEmail(), baseUrl: TF_CONFIG.pagesUrl || ScriptApp.getService().getUrl(),
    contact: TF_CONFIG.contact, uid: function () { return Utilities.getUuid().replace(/-/g, '').slice(0, 12); }, token: function () { return tfToken_(10); } };
}
function tfToken_(len) {
  var a = 'ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnpqrstuvwxyz23456789', s = '';
  var b = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, Utilities.getUuid() + Math.random() + Date.now());
  for (var i = 0; i < (len || 24); i++) s += a.charAt((b[i] & 255) % a.length);
  return s;
}
function tfRead_(ss, n) {
  var sh = ss.getSheetByName(n), cols = TF_TABLES[n], v = sh.getLastRow() > 1 ? sh.getRange(2, 1, sh.getLastRow() - 1, cols.length).getDisplayValues() : [];
  return v.filter(function (r) { return r[0] !== ''; }).map(function (r) {
    var o = {}; cols.forEach(function (c, i) { o[c] = TF_BOOL[c] ? r[i] === '1' : TF_LIST[c] ? (r[i] ? r[i].split(',') : []) : r[i]; }); return o;
  });
}
function tfSave_(ss, n, rows) {
  var sh = ss.getSheetByName(n), cols = TF_TABLES[n];
  if (sh.getLastRow() > 1) sh.getRange(2, 1, sh.getLastRow() - 1, cols.length).clearContent();
  if (!rows.length) return;
  var v = rows.map(function (o) { return cols.map(function (c) { var x = o[c]; return TF_BOOL[c] ? (x ? '1' : '0') : TF_LIST[c] ? (x || []).join(',') : String(x == null ? '' : x); }); });
  sh.getRange(2, 1, v.length, cols.length).setNumberFormat('@').setValues(v);
}
function tfLoad_(ss, P) {
  return { clients: tfRead_(ss, 'clients'), demandes: tfRead_(ss, 'demandes'), blocages: tfRead_(ss, 'blocages'),
    busy: tfRead_(ss, 'agenda').map(function (r) { return r.jour; }), notes: tfRead_(ss, 'notes'), bilans: tfRead_(ss, 'bilans'), attentes: tfRead_(ss, 'attentes'), evenements: tfRead_(ss, 'evenements'), ouverts: tfRead_(ss, 'ouverts').map(function (r) { return r.jour; }),
    meta: { lastSync: P.getProperty('LAST_SYNC') || '', syncError: P.getProperty('SYNC_ERROR') || '', agendas: P.getProperty('AGENDAS') || '', backup: P.getProperty('LAST_BACKUP') || '', mensuel: P.getProperty('LAST_MENSUEL') || '' } };
}

/* ---------- Synchronisation Google Agenda (toutes les 15 min + après chaque validation) ----------
   - Écrit les formations validées dans l'agenda du compte pro.
   - Lit TOUS les agendas du compte (pro + agendas perso partagés) : un événement « occupé » rend le jour
     indisponible pour les clients, sans jamais montrer le titre.
   - Maintenance : options expirées libérées, rappels, recyclage, demandes passées.
   Si Google Agenda ne répond pas : l'agenda continue sur le dernier calendrier connu et les opérations
   en attente sont rejouées automatiquement au passage suivant. */
function tfSync() {
  var P = PropertiesService.getScriptProperties(), lock = LockService.getScriptLock();
  if (!lock.tryLock(20000)) return;
  try {
    var ss = SpreadsheetApp.openById(P.getProperty('SS_ID')), cal = CalendarApp.getDefaultCalendar();
    tfEnsureSchema_(ss, P);
    try { tfBackup_(ss, P); } catch (x) { P.setProperty('BACKUP_ERROR', String(x && x.message || x)); }
    try { // maintenance automatique (options expirées, rappels…)
      var env = tfEnv_(P), mdb = tfLoad_(ss, P), m = tfApi(mdb, env, 'maintenance', env.adminKey, {});
      if (m.ok) { if (m.dirty.demandes) tfSave_(ss, 'demandes', mdb.demandes); if (m.dirty.attentes) tfSave_(ss, 'attentes', mdb.attentes); if (m.mails.length) tfQueueMails_(ss, m.mails); }
    } catch (x) { }
    var dem = tfRead_(ss, 'demandes'), clients = tfRead_(ss, 'clients'), changed = false, err = '';
    function d0(s) { var a = s.split('-'); return new Date(Number(a[0]), Number(a[1]) - 1, Number(a[2])); }
    dem.forEach(function (d) {
      try {
        if (d.sync === 'A_FAIRE') {
          var nom = '—'; clients.forEach(function (c) { if (c.id === d.clientId) nom = c.nom; });
          var ids = [], a = d.jours.slice().sort(), i = 0;
          var pres = d.test && !d.theorie && !d.pratique ? 'Test' : d.test ? 'Formation + test' : 'Formation';
          var desc = [d.formation, [d.theorie ? 'Théorie' : '', d.pratique ? 'Pratique' : '', d.test ? 'Test' : ''].filter(String).join(' + '),
            d.categories ? 'Catégories : ' + d.categories.split(',').join(', ') : '', Number(d.stagiaires) ? 'Stagiaires : ' + d.stagiaires : '',
            d.lieu ? 'Lieu : ' + d.lieu : '', d.horaires ? 'Horaires : ' + d.horaires.replace('-', ' – ') : '', d.contactSite ? 'Contact sur place : ' + d.contactSite : '', d.remarque].filter(String).join('\n');
          while (i < a.length) {
            var j = i; while (j + 1 < a.length && tfAddDays(a[j], 1) === a[j + 1]) j++;
            var ev = cal.createAllDayEvent(pres + ' ' + d.formation.split(' — ')[0] + ' — ' + nom, d0(a[i]), d0(tfAddDays(a[j], 1)), { description: desc, location: d.lieu || '' });
            ev.setTag('tf', d.id); try { ev.setTransparency(CalendarApp.EventTransparency.OPAQUE); } catch (x) { }
            ids.push(ev.getId()); i = j + 1;
          }
          d.eventId = ids.join('|'); d.sync = 'OK'; changed = true;
        } else if (d.sync === 'A_SUPPRIMER') {
          (d.eventId || '').split('|').filter(String).forEach(function (id) { var ev = cal.getEventById(id); if (ev) ev.deleteEvent(); });
          d.eventId = ''; d.sync = 'OK'; changed = true;
        }
      } catch (e) { err = String(e && e.message || e); }
    });
    var keep = dem.filter(function (d) { return !(d.statut === 'SUPPRIMEE' && d.sync !== 'A_SUPPRIMER'); }); // entreprises supprimées : on purge
    if (keep.length !== dem.length) { dem = keep; changed = true; }
    if (changed) tfSave_(ss, 'demandes', dem);

    var now = new Date(), fin = new Date(now.getTime() + (TF_CONFIG.horizon + 2) * 86400000), busy = {}, tz = TF_CONFIG.tz, used = [], cals = [], evts = [];
    try { cals = CalendarApp.getAllCalendars(); } catch (e) { cals = [cal]; }
    cals.forEach(function (c) {
      var nm = ''; try { nm = c.getName() || ''; } catch (x) { }
      if (TF_CONFIG.ignorer.test(nm)) return;
      try {
        c.getEvents(new Date(now.getTime() - 60 * 86400000), fin).forEach(function (ev) {
          if (ev.getTag('tf')) return;
          try { if (TF_CONFIG.ignorerEvt.test(ev.getTitle() || '')) return; } catch (x) { }
          try { if (ev.getMyStatus() === CalendarApp.GuestStatus.NO) return; } catch (x) { }
          var blk = true; // une journée entière bloque toujours (Google la marque « disponible » par défaut)
          try { if (!ev.isAllDayEvent() && ev.getTransparency() === CalendarApp.EventTransparency.TRANSPARENT) blk = false; } catch (x) { }
          try { var rec = tfEvtRec_(ev, c, nm, blk); if (rec) evts.push(rec); } catch (x) { }
          if (!blk) return;
          if (ev.isAllDayEvent()) {
            var s = Utilities.formatDate(ev.getAllDayStartDate(), Session.getScriptTimeZone(), 'yyyy-MM-dd'), e = Utilities.formatDate(ev.getAllDayEndDate(), Session.getScriptTimeZone(), 'yyyy-MM-dd');
            for (var x = s, n = 0; x < e && n < 400; x = tfAddDays(x, 1), n++) busy[x] = 1;
          } else {
            var S = Utilities.formatDate(ev.getStartTime(), tz, 'yyyy-MM-dd HH:mm'), E = Utilities.formatDate(ev.getEndTime(), tz, 'yyyy-MM-dd HH:mm');
            for (var y = S.slice(0, 10), k = 0; y <= E.slice(0, 10) && k < 400; y = tfAddDays(y, 1), k++)
              if (S < y + ' ' + TF_CONFIG.heureFin && E > y + ' ' + TF_CONFIG.heureDebut) busy[y] = 1;
          }
        });
        used.push(nm);
      } catch (e) { err = String(e && e.message || e); }
    });
    tfSave_(ss, 'agenda', Object.keys(busy).sort().map(function (j) { return { jour: j }; }));
    if (!err) { evts.sort(function (a, b) { return a.debut < b.debut ? -1 : a.debut > b.debut ? 1 : 0; }); tfSave_(ss, 'evenements', evts.slice(0, 2500)); }
    try { tfAlertes_(P, dem, clients, evts); } catch (x) { }
    P.setProperty('AGENDAS', used.join(' · '));
    P.setProperty('LAST_SYNC', new Date().toISOString()); P.setProperty('SYNC_ERROR', err);
  } catch (e) {
    P.setProperty('SYNC_ERROR', String(e && e.message || e));
  } finally { SpreadsheetApp.flush(); lock.releaseLock(); }
  try { tfAuto_(P); } catch (e) { P.setProperty('AUTO_ERROR', String(e && e.message || e)); }
  try { tfFlushMails_(); } catch (e) { }
}

/* ---------- Événements Google : fiche détaillée (formateur seulement, jamais montrée aux clients) ---------- */
function tfEvtRec_(ev, c, nm, blk) {
  var tz = TF_CONFIG.tz, ad = ev.isAllDayEvent(), st = ev.getStartTime(), o = { k: c.getId() + '|' + ev.getId() + '|' + st.toISOString(), cal: nm, titre: ev.getTitle() || '(sans titre)', tout: ad,
    lieu: ev.getLocation() || '', desc: String(ev.getDescription() || '').slice(0, 1500), bloque: blk, rec: false, own: false };
  try { o.rec = ev.isRecurringEvent(); } catch (x) { }
  try { o.own = c.isOwnedByMe(); } catch (x) { }
  if (ad) {
    var s = Utilities.formatDate(ev.getAllDayStartDate(), Session.getScriptTimeZone(), 'yyyy-MM-dd'), e = Utilities.formatDate(ev.getAllDayEndDate(), Session.getScriptTimeZone(), 'yyyy-MM-dd');
    o.debut = s; o.fin = e > s ? tfAddDays(e, -1) : s;
  } else { o.debut = Utilities.formatDate(st, tz, 'yyyy-MM-dd HH:mm'); o.fin = Utilities.formatDate(ev.getEndTime(), tz, 'yyyy-MM-dd HH:mm'); }
  return o;
}
function tfD0_(s) { var a = s.split('-'); return new Date(Number(a[0]), Number(a[1]) - 1, Number(a[2])); }
function tfEvtFind_(k) {
  var a = String(k || '').split('|'); if (a.length < 3) throw new Error('Événement introuvable.');
  var cal = CalendarApp.getCalendarById(a[0]); if (!cal) throw new Error('Agenda introuvable.');
  var t = new Date(a[2]).getTime(), f = null;
  cal.getEvents(new Date(t - 3600000), new Date(t + 3600000)).forEach(function (e) { if (!f && e.getId() === a[1] && Math.abs(e.getStartTime().getTime() - t) < 60000) f = e; });
  if (!f) throw new Error('Cet événement a été modifié dans Google Agenda. La liste vient d\'être actualisée, réessayez.');
  return f;
}
function tfEvt_(action, p) {
  var titre = String(p.titre || '').replace(/\s+/g, ' ').trim().slice(0, 200), lieu = String(p.lieu || '').trim().slice(0, 200), desc = String(p.desc == null ? '' : p.desc).replace(/\r/g, '').slice(0, 4000);
  if (action !== 'evtSuppr' && !titre) throw new Error('Donnez un titre à l\'événement.');
  var fin = '', jour = String(p.jour || '');
  if (action !== 'evtSuppr' && (action === 'evtCreer' || p.tps)) {
    if (!tfIsDay(jour)) throw new Error('Date de début invalide.');
    fin = tfIsDay(p.jourFin) ? p.jourFin : jour; if (fin < jour) throw new Error('La date de fin est avant la date de début.');
    if (!p.tout) {
      if (!/^\d{2}:\d{2}$/.test(p.hd || '') || !/^\d{2}:\d{2}$/.test(p.hf || '')) throw new Error('Indiquez les heures de début et de fin.');
      if (fin + ' ' + p.hf <= jour + ' ' + p.hd) throw new Error('L\'heure de fin doit être après l\'heure de début.');
    }
  }
  function dt(j, h) { return Utilities.parseDate(j + ' ' + h, TF_CONFIG.tz, 'yyyy-MM-dd HH:mm'); }
  try {
    if (action === 'evtCreer') {
      var cal = CalendarApp.getDefaultCalendar(), ev;
      if (p.tout) ev = cal.createAllDayEvent(titre, tfD0_(jour), tfD0_(tfAddDays(fin, 1)), { description: desc, location: lieu });
      else ev = cal.createEvent(titre, dt(jour, p.hd), dt(fin, p.hf), { description: desc, location: lieu });
      try { ev.setTransparency(CalendarApp.EventTransparency.OPAQUE); } catch (x) { }
      return 'Événement ajouté à Google Agenda';
    }
    var e = tfEvtFind_(p.k);
    if (action === 'evtSuppr') { e.deleteEvent(); return 'Événement supprimé de Google Agenda'; }
    e.setTitle(titre); e.setLocation(lieu); e.setDescription(desc);
    if (p.tps) { if (p.tout) e.setAllDayDates(tfD0_(jour), tfD0_(tfAddDays(fin, 1))); else e.setTime(dt(jour, p.hd), dt(fin, p.hf)); }
    return 'Modifié dans Google Agenda';
  } catch (x) {
    var m = String(x && x.message || x);
    if (/Accès refusé|permission|autorisation|not (have|allowed)|read.?only|forbidden/i.test(m)) throw new Error('Google refuse la modification : cet agenda est en lecture seule. Partagez-le avec le droit « Apporter des modifications aux événements ».');
    throw new Error(m);
  }
}

/* ---------- Alerte : un nouvel événement Google tombe sur une formation validée ---------- */
function tfAlertes_(P, dem, clients, evts) {
  var today = Utilities.formatDate(new Date(), TF_CONFIG.tz, 'yyyy-MM-dd'), cur = {}, lines = [], first = P.getProperty('ALERTES') === null;
  var seen = {}; try { (JSON.parse(P.getProperty('ALERTES') || '[]')).forEach(function (k) { seen[k] = 1; }); } catch (x) { }
  dem.forEach(function (d) {
    if (d.statut !== 'VALIDEE') return;
    evts.forEach(function (e) {
      if (!e.bloque) return;
      var a = e.debut.slice(0, 10), b = e.fin.slice(0, 10), hit = null;
      d.jours.forEach(function (j) { if (j >= today && j >= a && j <= b && !hit) hit = j; });
      if (!hit) return;
      var key = d.id + '|' + e.k; cur[key] = 1;
      if (!seen[key] && !first) { var nom = '—'; clients.forEach(function (c) { if (c.id === d.clientId) nom = c.nom; }); lines.push('« ' + e.titre + ' » (' + e.cal + ') le ' + tfFmtJours([hit]) + ' — formation ' + d.formation + ' chez ' + nom); }
    });
  });
  if (lines.length) {
    var env = tfEnv_(P), ss = SpreadsheetApp.openById(P.getProperty('SS_ID'));
    tfQueueMails_(ss, [{ to: env.adminEmail, sujet: 'Conflit d\'agenda détecté', titre: 'Conflit d\'agenda', lignes: ['Un événement Google Agenda vient d\'être ajouté sur une formation déjà validée :'].concat(lines).concat(['Ouvrez votre espace pour déplacer ou ajuster l\'un des deux.']), lien: { label: 'Ouvrir mon espace formateur', url: env.baseUrl + '?admin=' + env.adminKey } }]);
  }
  P.setProperty('ALERTES', JSON.stringify(Object.keys(cur).slice(-400)));
}

/* ---------- Tâches automatiques : e-mail mensuel le 1er, résumé du lundi (à partir de 7 h) ---------- */
function tfAuto_(P) {
  var now = new Date(), tz = TF_CONFIG.tz;
  if (Number(Utilities.formatDate(now, tz, 'H')) < 7) return;
  var today = Utilities.formatDate(now, tz, 'yyyy-MM-dd'), ym = today.slice(0, 7), dom = Number(today.slice(8, 10)), dow = Number(Utilities.formatDate(now, tz, 'u'));
  var needM = dom === 1 && P.getProperty('LAST_MENSUEL') !== ym, needD = dow === 1 && P.getProperty('LAST_DIGEST') !== today;
  if (!needM && !needD) return;
  var lock = LockService.getScriptLock(); if (!lock.tryLock(20000)) return;
  try {
    var ss = SpreadsheetApp.openById(P.getProperty('SS_ID')), env = tfEnv_(P), db = tfLoad_(ss, P), r;
    if (needM) { r = tfApi(db, env, 'mensuel', env.adminKey, { mode: 'envoi', ym: ym }); if (r.ok) { if (r.mails.length) tfQueueMails_(ss, r.mails); P.setProperty('LAST_MENSUEL', ym); P.setProperty('LAST_MENSUEL_N', String(r.mails.length)); } }
    if (needD) { r = tfApi(db, env, 'digest', env.adminKey, {}); if (r.ok) { if (r.mails.length) tfQueueMails_(ss, r.mails); P.setProperty('LAST_DIGEST', today); } }
  } finally { lock.releaseLock(); }
}

/* ---------- Sauvegarde quotidienne (dans votre Drive : classeur « Sauvegardes agenda », 14 derniers jours) ----------
   Copie clients, demandes, blocages et jours ouverts. Ne touche jamais aux données d'origine. */
function tfBackup_(ss, P) {
  var day = Utilities.formatDate(new Date(), TF_CONFIG.tz, 'yyyy-MM-dd');
  if (P.getProperty('BACKUP_DAY') === day) return;
  var id = P.getProperty('BACKUP_SS_ID'), bk = null;
  if (id) { try { bk = SpreadsheetApp.openById(id); } catch (e) { bk = null; } }
  if (!bk) { bk = SpreadsheetApp.create('Torres Formation — Sauvegardes agenda (ne pas supprimer)'); P.setProperty('BACKUP_SS_ID', bk.getId()); }
  ['clients', 'demandes', 'blocages', 'ouverts', 'notes', 'bilans', 'attentes'].forEach(function (n) {
    var src = ss.getSheetByName(n); if (!src) return;
    var nm = day + ' · ' + n, old = bk.getSheetByName(nm);
    if (old) bk.deleteSheet(old);
    src.copyTo(bk).setName(nm);
  });
  var lim = Utilities.formatDate(new Date(Date.now() - 14 * 86400000), TF_CONFIG.tz, 'yyyy-MM-dd');
  bk.getSheets().forEach(function (sh) {
    var m = sh.getName().match(/^(\d{4}-\d{2}-\d{2}) · /);
    if (m && m[1] < lim) bk.deleteSheet(sh);
  });
  bk.getSheets().forEach(function (sh) { if (/^(Sheet1|Feuille 1)$/.test(sh.getName()) && bk.getSheets().length > 1) bk.deleteSheet(sh); });
  P.setProperty('BACKUP_DAY', day); P.setProperty('LAST_BACKUP', new Date().toISOString()); P.deleteProperty('BACKUP_ERROR');
}

  tfEnsureSchema_(ss, P);
  var db = tfLoad_(ss, P), ko = [], ok = 0, actifs = 0;

    if (r.ok && r.data && r.data.role === 'client') ok++; else ko.push(c.nom);

function tfFmtJours(jours) {
  var J = ['dim.', 'lun.', 'mar.', 'mer.', 'jeu.', 'ven.', 'sam.'];
  var M = ['janv.', 'févr.', 'mars', 'avr.', 'mai', 'juin', 'juil.', 'août', 'sept.', 'oct.', 'nov.', 'déc.'];
  function f(s) { var d = new Date(s + 'T12:00:00Z'); return J[d.getUTCDay()] + ' ' + d.getUTCDate() + ' ' + M[d.getUTCMonth()] + ' ' + d.getUTCFullYear(); }
  var a = jours.slice().sort(), out = [], i = 0;
  while (i < a.length) {
    var j = i; while (j + 1 < a.length && tfAddDays(a[j], 1) === a[j + 1]) j++;
    out.push(j > i ? 'du ' + f(a[i]) + ' au ' + f(a[j]) : f(a[i])); i = j + 1;
  }
  return out.join(', ');
}

function tfPaques(y) {
  var a = y % 19, b = Math.floor(y / 100), c = y % 100, d = Math.floor(b / 4), e = b % 4, f = Math.floor((b + 8) / 25), g = Math.floor((b - f + 1) / 3),
    h = (19 * a + b - d - g + 15) % 30, i = Math.floor(c / 4), k = c % 4, l = (32 + 2 * e + 2 * i - h - k) % 7, m = Math.floor((a + 11 * h + 22 * l) / 451),
    mo = Math.floor((h + l - 7 * m + 114) / 31), da = ((h + l - 7 * m + 114) % 31) + 1;
  return y + '-' + (mo < 10 ? '0' : '') + mo + '-' + (da < 10 ? '0' : '') + da;
}

}

      var l = ['Formation : ' + d.formation + (prestaLabel(d) ? ' (' + prestaLabel(d) + ')' : '')];
      if (d.categories) l.push('Catégories : ' + d.categories.split(',').join(', '));
      l.push('Dates : ' + tfFmtJours(d.jours));
      if (nb(d)) l.push('Stagiaires : ' + nb(d) + (TF_PUBLICS[d.public] ? ' — ' + TF_PUBLICS[d.public].toLowerCase() : ''));
      else if (TF_PUBLICS[d.public]) l.push('Public : ' + TF_PUBLICS[d.public].toLowerCase());
      l.push('Type : ' + (d.type === 'FERME' ? 'réservation ferme' : 'à confirmer par l\'entreprise'));
      if (d.lieu) l.push('Lieu : ' + d.lieu);
      if (d.horaires) l.push('Horaires : ' + d.horaires.replace('-', ' – '));
      if (d.contactSite) l.push('Contact sur place : ' + d.contactSite);
      if (d.remarque) l.push('Remarque : ' + d.remarque);
      return l;
    }
    function mailAdmin(sujet, titre, lignes) {
      out.mails.push({ to: env.adminEmail, sujet: sujet, titre: titre, lignes: lignes, lien: { label: 'Ouvrir mon espace formateur', url: env.baseUrl + '?admin=' + env.adminKey } });
    }
    function mailClient(c, sujet, titre, lignes, d) { // d (facultatif) : la demande concernée → boutons d'action directs dans l'e-mail
      if (!c || !c.email) return;
      var m = { to: c.email, sujet: sujet, titre: titre, lignes: lignes, lien: { label: 'Ouvrir mon agenda', url: env.baseUrl + '?c=' + c.token } };
      if (d && (d.statut === 'ATTENTE' || d.statut === 'VALIDEE')) {
        var u = env.baseUrl + '?c=' + c.token + '&d=' + d.id + '&x=', a = [];
        if (d.statut === 'ATTENTE' && d.type === 'OPTION') a.push({ label: 'Confirmer', url: u + 'confirmer' });
        if (!d.remplace) a.push({ label: d.statut === 'VALIDEE' ? 'Demander un déplacement' : 'Déplacer', url: u + 'deplacer' });
        if (d.statut === 'ATTENTE') a.push({ label: 'Annuler', url: u + 'annuler' });
        if (a.length) m.actions = a;
      }
      out.mails.push(m);
    }
    /* Exécute un changement qui libère des dates ; prévient les entreprises en file d'attente qui passent en tête */
    function liberer(d, change) {
      var avant = {};
      db.demandes.forEach(function (o) { if (o.id !== d.id && o.statut === 'ATTENTE' && partage(d, o)) avant[o.id] = rang(o).rang; });
      change();
      db.demandes.forEach(function (o) {
        if (avant[o.id] > 1 && o.statut === 'ATTENTE' && rang(o).rang === 1)
          mailClient(clientById(o.clientId), 'Votre option est passée en tête', 'Une date s\'est libérée', ['Bonjour,', 'Une date que vous aviez demandée s\'est libérée : votre demande est maintenant la première de la file.', 'Dates : ' + tfFmtJours(o.jours)].concat(o.type === 'OPTION' ? ['Pensez à confirmer votre formation pour la sécuriser.'] : []), o);
      });
    }
    function normJours(a) { return Array.isArray(a) ? a.filter(function (j, i, x) { return x.indexOf(j) === i; }).sort() : []; }
    function fiche(q, strict) {
      if (TF_FORMATIONS.indexOf(q.formation) < 0) fail('Choisissez une formation.');
      var th = !!q.theorie, pr = !!q.pratique, te = !!q.test;
      if (te && !TF_TEST[q.formation]) fail('Le test n\'est pas proposé pour cette formation.');
      if (!th && !pr && !te) fail('Choisissez théorie, pratique ou test.');
      var list = TF_CATS[q.formation] || [], cats = (Array.isArray(q.categories) ? q.categories : []).filter(function (c, i, a) { return list.indexOf(c) >= 0 && a.indexOf(c) === i; });
      var n = parseInt(q.stagiaires, 10); if (isNaN(n) || n < 0) n = 0; if (n > 60) n = 60;
      var pub = TF_PUBLICS[q.public] ? q.public : '';
      var hr = /^([01]\d|2[0-3]):[0-5]\d-([01]\d|2[0-3]):[0-5]\d$/.test(String(q.horaires || '')) ? String(q.horaires) : '';
      return { horaires: hr, formation: q.formation, theorie: th, pratique: pr, test: te, categories: cats.join(','), stagiaires: n, public: pub,
        lieu: tfStr(q.lieu, 160), contactSite: tfStr(q.contactSite, 120), remarque: tfStr(q.remarque, 600) };
    }
    function nouvelle(f, clientId, jours, type) {
      var nd = { id: env.uid(), clientId: clientId, jours: jours, formation: f.formation, theorie: f.theorie, pratique: f.pratique, test: f.test, type: type,
        lieu: f.lieu, remarque: f.remarque, statut: 'ATTENTE', creeLe: env.now, majLe: env.now, eventId: '', sync: '', stagiaires: f.stagiaires, public: f.public,
        categories: f.categories, contactSite: f.contactSite, expireLe: '', source: 'CLIENT', rappelLe: '', recycleLe: '', remplace: '', rappelFormation: '', horaires: f.horaires || '', histo: '' };
      nd.rappelLe = rap0(nd); return nd;
    }
    function checkJours(jours, clientId, ignore) {
      if (!jours.length) fail('Sélectionnez au moins un jour.');
      if (jours.length > 31) fail('31 jours maximum par demande.');
      jours.forEach(function (j) {
        if (!tfIsDay(j) || j < minDay || j > maxDay) fail(tfIsDay(j) && j <= env.today ? 'Pour une formation aujourd\'hui ou en urgence, appelez directement Torres Formation.' : 'Une des dates n\'est pas réservable.');
        var e = etatJour(j, clientId); if (e && e.s === 'x') fail('Une des dates vient d\'être prise. Actualisez et choisissez une autre date.');
        db.demandes.forEach(function (d) { if (d.clientId === clientId && actif(d) && ignore.indexOf(d.id) < 0 && d.jours.indexOf(j) >= 0) fail('Vous avez déjà une demande sur une de ces dates.'); });
      });
    }

    function doValider(v, silent) {
      if (v.statut !== 'ATTENTE') fail('Cette demande n\'est plus en attente.');
      db.demandes.forEach(function (o) { if (o.id !== v.id && o.id !== v.remplace && o.statut === 'VALIDEE' && partage(v, o)) fail('Une formation est déjà validée sur une de ces dates (' + clientNom(o.clientId) + ').'); });
      v.statut = 'VALIDEE'; v.type = 'FERME'; v.expireLe = ''; v.majLe = env.now; v.sync = 'A_FAIRE'; histo(v, 'Validée');
      if (!silent) mailClient(clientById(v.clientId), 'Votre formation est validée', 'Formation validée', ['Bonjour,', 'Bonne nouvelle : votre formation est validée.'].concat(resume(v)).concat(['Je reviendrai vers vous pour les détails pratiques.']), v);
      if (v.remplace) {
        var old = demById(v.remplace);
        if (old && old.statut === 'VALIDEE') { old.statut = 'ANNULEE'; old.majLe = env.now; histo(old, 'Remplacée par un déplacement'); old.sync = old.eventId ? 'A_SUPPRIMER' : ''; }
      }
      db.demandes.forEach(function (o) {
        if (o.id !== v.id && o.statut === 'ATTENTE' && partage(v, o)) {
          o.statut = 'REFUSEE'; o.majLe = env.now; histo(o, 'Non retenue : date validée pour une autre demande');
          if (o.clientId === v.clientId) { o.statut = 'ANNULEE'; return; }
          mailClient(clientById(o.clientId), 'Date indisponible', 'Date non retenue', ['Bonjour,', 'La date demandée n\'est finalement plus disponible.', 'Dates : ' + tfFmtJours(o.jours), 'Vous pouvez choisir une autre date dans votre agenda.']);
        }
      });
    }

    if (action === 'noteMaj') { // note privée sur une date (vide = supprimée)
      if (!tfIsDay(p.jour)) fail('Date invalide.');

      if (tn) db.notes.push({ jour: p.jour, texte: tn, majLe: env.now });

    }
    if (action === 'bilanJour') { // bilan privé d'une journée : prix facturé, stagiaires, texte (tout vide = supprimé)
      if (!tfIsDay(p.jour)) fail('Date invalide.');
      var pr = String(p.prix == null ? '' : p.prix).replace(/\s|€/g, '').replace(',', '.'), st = String(p.stagiaires == null ? '' : p.stagiaires).replace(/\s/g, '');
      if (pr !== '') { if (!/^\d{1,6}(\.\d{1,2})?$/.test(pr)) fail('Prix invalide : écrivez un montant, par exemple 350 ou 420,50.'); pr = String(Number(pr)); }

      var tb = String(p.texte == null ? '' : p.texte).replace(/\r/g, '').trim().slice(0, 1000);
      db.bilans = (db.bilans || []).filter(function (b) { return b.jour !== p.jour; });
      if (pr !== '' || st !== '' || tb) db.bilans.push({ jour: p.jour, prix: pr, stagiaires: st, texte: tb, majLe: env.now });
      out.dirty.bilans = true; out.data = initAdmin(); out.data.msg = (pr !== '' || st !== '' || tb) ? 'Bilan de la journée enregistré' : 'Bilan supprimé'; return out;

    if (action === 'clientMensuel') {
      var cm = clientById(p.id); if (!cm) fail('Entreprise introuvable.');
      cm.mensuel = p.on ? '1' : '0'; out.dirty.clients = true; out.data = initAdmin(); out.data.msg = p.on ? 'E-mail mensuel activé pour ' + cm.nom : 'E-mail mensuel désactivé pour ' + cm.nom; return out;
    }
    /* ---- e-mail mensuel des disponibilités ---- */
    function dispoMois(ym, cid) {
      var libres = [], opts = [], cl = null, co = null, nb = 0, no = 0;
      for (var j = ym + '-01'; j.slice(0, 7) === ym; j = tfAddDays(j, 1)) {
        if (nat(j)) continue; // l'e-mail liste les jours ouvrés ; week-ends et fériés restent réservables dans l'agenda
        var st = 'x';
        if (j >= minDay && j <= maxDay && !hard[j]) { var e = etatJour(j, cid); st = !e ? 'l' : e.s === 'o' ? 'o' : 'x'; }
        if (st === 'l') { nb++; if (cl) { cl[1] = j; cl[2]++; } else { cl = [j, j, 1]; libres.push(cl); } co = null; }
        else if (st === 'o') { no++; if (co) { co[1] = j; co[2]++; } else { co = [j, j, 1]; opts.push(co); } cl = null; }
        else { cl = null; co = null; }
      }
      return { libres: libres, opts: opts, nb: nb, no: no };
    }
    function mensuelMail(c, ym, skipEmpty) {
      var dm = dispoMois(ym, c.id);
      if (skipEmpty && !dm.nb) return null;
      var an = ym.slice(0, 4), mois = TF_MOIS_LONG[+ym.slice(5, 7) - 1], ny = ym.slice(5, 7) === '12' ? (+an + 1) + '-01' : an + '-' + ('0' + (+ym.slice(5, 7) + 1)).slice(-2);
      var nx = dispoMois(ny, c.id), nxMois = TF_MOIS_LONG[+ny.slice(5, 7) - 1];
      var td = 'padding:11px 0;border-bottom:1px solid #EDEAE4;';
      function rows(a, grey) { return a.map(function (r) { return '<tr><td style="' + td + 'font-size:15px;color:' + (grey ? '#6E6B66' : '#0B0B0C') + ';">' + (grey ? '' : '<b>') + tfEscH(tfFmtPlage(r[0], r[1])) + (grey ? '' : '</b>') + '</td><td align="right" style="' + td + 'font-size:13px;color:#6E6B66;white-space:nowrap;">' + r[2] + ' jour' + (r[2] > 1 ? 's' : '') + '</td></tr>'; }).join(''); }
      var h = '';
      if (dm.nb) h += '<div style="margin:20px 0 8px;padding:14px 16px;background:#EDEAE4;border-left:3px solid #C22A22;font-size:15px;color:#0B0B0C;"><span style="font-size:28px;font-weight:700;letter-spacing:1px;">' + dm.nb + '</span> jour' + (dm.nb > 1 ? 's' : '') + ' disponible' + (dm.nb > 1 ? 's' : '') + ' en ' + mois + '</div><table role="presentation" width="100%" style="border-collapse:collapse;margin:6px 0 4px;">' + rows(dm.libres, false) + '</table>';
      if (dm.opts.length) h += '<p style="margin:18px 0 4px;font-size:12px;letter-spacing:2px;text-transform:uppercase;color:#6E6B66;">Sous option par une autre entreprise — peuvent se libérer</p><table role="presentation" width="100%" style="border-collapse:collapse;">' + rows(dm.opts, true) + '</table>';
      var l = ['Bonjour,', dm.nb ? 'Le mois de ' + mois + ' est entamé : voici mes disponibilités pour vos prochaines formations.' : 'Il ne reste plus de date disponible en ' + mois + ' ' + an + '. Les mois suivants sont ouverts à la réservation dans votre agenda personnel.'];
      var ex = [];
      if (nx.nb) ex.push('Et en ' + nxMois + ' : ' + nx.nb + ' jour' + (nx.nb > 1 ? 's' : '') + ' encore disponible' + (nx.nb > 1 ? 's' : '') + '.');
      ex.push('Vous pouvez réserver via ce lien directement. Je reste disponible pour toute question.');
      h += ex.map(function (t) { return '<p style="margin:14px 0 0;font-size:15px;line-height:1.5;color:#17171A;">' + tfEscH(t) + '</p>'; }).join('');
      return { to: c.email, sujet: 'Mes disponibilités de ' + mois + ' ' + an, titre: 'Disponibilités de ' + mois + ' ' + an, lignes: l, bloc: h, fin: ['Au plaisir de contribuer à la sécurité avec vous,', 'Maïcol Torres'],
        lien: { label: 'Consulter l\'agenda et réserver', url: env.baseUrl + '?c=' + c.token },
        pied: 'Message automatique envoyé le 1er de chaque mois aux entreprises clientes. <a href="' + tfEscH(env.baseUrl + '?c=' + c.token + '&m=stop') + '" style="color:#6E6B66;">Ne plus recevoir cet e-mail</a>' };
    }
    if (action === 'mensuel') {
      var ym = /^\d{4}-(0[1-9]|1[0-2])$/.test(p.ym || '') ? p.ym : env.today.slice(0, 7);
      var cibles = db.clients.filter(function (c) { return c.actif && c.email && c.mensuel !== '0'; }), nm2 = 0;
      if (p.mode === 'apercu') {
        var c0 = (p.id && clientById(p.id)) || cibles[0] || db.clients[0]; if (!c0) fail('Créez d\'abord une entreprise.');
        var ma = mensuelMail(c0, ym, false); ma.to = env.adminEmail; ma.sujet = '[Aperçu] ' + ma.sujet; ma.pied = 'APERÇU — exemple avec le lien de « ' + c0.nom + ' ». ' + ma.pied;
        out.mails.push(ma); out.data = initAdmin(); out.data.msg = 'Aperçu envoyé à ' + env.adminEmail; return out;
      }
      cibles.forEach(function (c) { var m2 = mensuelMail(c, ym, true); if (m2) { out.mails.push(m2); nm2++; } });
      out.data = initAdmin(); out.data.envoyes = nm2; out.data.msg = nm2 ? 'E-mail envoyé à ' + plurielN(nm2, 'entreprise') : 'Aucun envoi : aucune entreprise avec e-mail et des dates libres'; return out;
    }
    function plurielN(n, w) { return n + ' ' + w + (n > 1 ? 's' : ''); }
    if (action === 'digest') { // résumé du lundi pour le formateur
      var lun = tfAddDays(env.today, -((new Date(env.today + 'T12:00:00Z').getUTCDay() + 6) % 7)), dim = tfAddDays(lun, 6), L = [];
      var fs = db.demandes.filter(function (d) { return d.statut === 'VALIDEE' && d.jours.some(function (j) { return j >= lun && j <= dim; }); }).sort(function (a, b) { return a.jours[0] < b.jours[0] ? -1 : 1; });
      var att = db.demandes.filter(function (d) { return d.statut === 'ATTENTE'; });
      var exp = att.filter(function (d) { return d.type === 'OPTION' && d.jours[0] <= tfAddDays(env.today, 14); });
      var libres7 = 0; for (var q = env.today; q <= dim; q = tfAddDays(q, 1)) if (!nat(q) && !hard[q] && q >= env.today && !db.demandes.some(function (d) { return d.statut === 'VALIDEE' && d.jours.indexOf(q) >= 0; })) libres7++;
      if (!fs.length && !att.length) { out.data = { vide: true }; return out; }
      L.push('Bonjour,', 'Voici votre semaine du ' + tfFmtJours([lun]) + ' au ' + tfFmtJours([dim]) + '.');
      L.push(fs.length ? 'Formations cette semaine (' + fs.length + ') :' : 'Aucune formation validée cette semaine.');
      fs.forEach(function (d) { L.push('• ' + clientNom(d.clientId) + ' — ' + d.formation + ' — ' + tfFmtJours(d.jours) + (d.lieu ? ' — ' + d.lieu : '')); });
      if (att.length) L.push(att.length + ' demande' + (att.length > 1 ? 's' : '') + ' en attente de votre réponse.');
      exp.forEach(function (d) { L.push('Non confirmé par ' + clientNom(d.clientId) + ' : ' + tfFmtJours(d.jours) + '.'); });
      L.push(libres7 + ' jour' + (libres7 > 1 ? 's' : '') + ' ouvré' + (libres7 > 1 ? 's' : '') + ' encore libre' + (libres7 > 1 ? 's' : '') + ' d\'ici dimanche.');
      mailAdmin('Votre semaine du ' + tfFmtJours([lun]).replace(/^\S+ /, ''), 'Résumé de la semaine', L);
      out.data = {}; return out;
    }
    if (action === 'maintenance') { // appelée toutes les 15 min par la synchronisation
      var ch = false;
      db.demandes.slice().forEach(function (d) {
        if (d.statut === 'ATTENTE') {
          if (d.jours[0] < env.today) { liberer(d, function () { d.statut = 'EXPIREE'; d.majLe = env.now; histo(d, 'Date passée sans validation'); }); ch = true; return; }
          if (d.type === 'OPTION') { // jamais d'annulation automatique : le formateur est simplement prévenu 14 puis 7 jours avant
            var lim = d.rappelLe !== '7' && d.jours[0] <= tfAddDays(env.today, 7) ? '7' : !d.rappelLe && d.jours[0] <= tfAddDays(env.today, 14) ? '14' : '';
            if (lim) {
              d.rappelLe = lim; ch = true; var nj = Math.round((new Date(d.jours[0] + 'T12:00:00Z') - new Date(env.today + 'T12:00:00Z')) / 86400000);
              mailAdmin(clientNom(d.clientId) + ' n\'a pas confirmé', 'Formation non confirmée', [clientNom(d.clientId) + ' n\'a pas encore confirmé sa formation, qui commence dans ' + nj + ' jour' + (nj > 1 ? 's' : '') + '.'].concat(resume(d)).concat(['La demande reste en attente : à vous de relancer l\'entreprise, de valider ou de refuser.']));
            }
          }
        } else if (d.statut === 'VALIDEE' && !d.recycleLe && /^(CACES|AIPR)/.test(d.formation)) {
          var fin = d.jours[d.jours.length - 1];
          if (fin < env.today && tfAddDays(env.today, -1700) >= fin) {
            d.recycleLe = env.today; ch = true;
            var echeance = (+fin.slice(0, 4) + 5) + fin.slice(4);
            var cx = clientById(d.clientId);
            mailClient(cx, 'Pensez au recyclage', 'Recyclage à prévoir', ['Bonjour,', 'La formation « ' + d.formation + ' » réalisée du ' + tfFmtJours(d.jours) + ' arrive à échéance vers le ' + tfFmtJours([echeance]) + '.', 'C\'est le bon moment pour planifier le recyclage de vos équipes.']);
            mailAdmin('Recyclage à proposer — ' + clientNom(d.clientId), 'Recyclage à proposer', ['Entreprise : ' + clientNom(d.clientId), 'Formation : ' + d.formation, 'Réalisée : ' + tfFmtJours(d.jours), 'Échéance vers le ' + tfFmtJours([echeance]), cx && cx.email ? 'Un rappel a été envoyé au client.' : 'Pas d\'e-mail enregistré : pensez à le contacter.']);
          }
        }
      });
      if (ch) out.dirty.demandes = true;
      out.data = {}; return out;
    }
    if (action === '__attentes') { // interne : une date surveillée s'est-elle libérée ?
      var keepA = [];
      db.attentes.forEach(function (a) {
        if (a.jour < env.today) { out.dirty.attentes = true; return; }
        var ca = clientById(a.clientId);
        if (!ca) { out.dirty.attentes = true; return; }
        if (a.jour >= minDay && !etatJour(a.jour, a.clientId)) {
          out.dirty.attentes = true;
          if (ca.actif && !db.demandes.some(function (d) { return d.clientId === a.clientId && actif(d) && d.jours.indexOf(a.jour) >= 0; }))
            mailClient(ca, 'Une date s\'est libérée', 'Date de nouveau disponible', ['Bonjour,', 'La date du ' + tfFmtJours([a.jour]) + ', pour laquelle vous aviez demandé à être prévenu, est de nouveau disponible.', 'Elle n\'est pas réservée pour vous : la première demande reçue est servie.']);
          return;
        }
        keepA.push(a);
      });
      db.attentes = keepA; out.data = {}; return out;
    }
    fail('Action inconnue.');
  } catch (e) {
    if (e && e.tf) return { ok: false, error: e.message, mails: [], dirty: {}, sync: false };
    throw e;
  }
